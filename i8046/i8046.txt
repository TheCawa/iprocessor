
Registers:
	00h - r0
	
	01h - X1	04h - X2	07h - X3	0Ah - X4	0Dh - X5
	02h - XL1	05h - XL2	08h - XL3	0Bh - XL4	0Eh - XL5
	03h - XH1	06h - XH2	09h - XH3	0Ch - XH4	0Fh - XH5
	
	10h - IX	14h - CS	18h - SCS	1Ch - A0
	11h - IY	15h - DS	19h - SDS	1Dh - A1
	12h - SP	16h - SS	1Ah - SSS	1Eh - FL
	13h - BP	17h - ES	1Bh - SES	1Fh - IC
	
	-# Xn = [XHn + XLn]
	-# Addr = [R + S] >= Size
	
Modes:
	#ID - #BIN - #NAME
	00h - 0000 - Byte
	01h - 0001 - Word 
	02h - 0010 - Addr
	03h - 0011 - Dword X
	04h - 0100 - Register (Lookup)
	05h - 0101 - Qword X
