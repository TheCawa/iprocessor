
Registers:
	00h - r0
	
	01h - EX1	02h - EX2	03h - EX3	04h - EX4	05h - EX5	06h - EX6	07h - EX7
	08h - X1	09h - X2	0Ah - X3	0Bh - X4	0Ch - X5	0Dh - X6	0Eh - X7
	0Fh - XL1	10h - XL2	11h - XL3	12h - XL4	13h - XL5	14h - XL6	15h - XL7
	
	16h - IC	19h - BP	1Ch - A0	1Fh - A3	22h - A6
	17h - FL	1Ah - IX	1Dh - A1	20h - A4	23h - A7
	18h - SP	1Bh - IY	1Eh - A2	21h - A5	
	
Modes:
	#ID - #BIN - #NAME
	00h - 0000 - Byte
	01h - 0001 - Word 
	02h - 0010 - Dword
	03h - 0011 - Qword X
	04h - 0100 - Register (Lookup)
	
Instruction types:
	#ID - #Name - #Length - #Mnemonic
	
	00h - S0    - 1B 	  - <OP>
	01h - S1	- 2B	  - <OP> <rD>
	02h - S2 	- 2B	  - <OP> <rS>
	03h - S3    - 2B      - <OP> <rS/D>
	04h - S4    - 2B      - <OP> <imm8>
	05h - S5    - 3B	  - <OP> <imm16>
	06h - S6 	- 4B	  - <OP> <imm24>
	07h - S7 	- 5B	  - <OP> <imm32>
	
	07h - A0	- 4B	  - <OP> <SF> <rS/D> <rS>
	08h - A1	- 4B	  - <OP> <SF> <rS/D> <imm8>
	09h - A2	- 5B	  - <OP> <SF> <rS/D> <imm16>
	0Ah - A3	- 7B	  - <OP> <SF> <rS/D> <imm32>
	
	0Bh - J0	- 6B	  - <OP> <COND> <imm32>
	0Ch - J1 	- 3B	  - <OP> <COND> <rS>
	0Dh - J2 	- 6B 	  - <OP> <COND> <disp32>
	
	0Eh - I0	- 6B	  - <OP> <imm8> <imm32>
	0Fh - H0 	- 3B	  - <OP> <rS/D> <disp8>
	
	10h - D0	- 4B	  - <OP> <SF> <rD> <rS>
	11h - D1B 	- 4B	  - <OP> <SF> <rD> <imm>
	12h - D1W	- 5B	  - <OP> <SF> <rD> <imm16>
	13h - D1DW  - 6B	  - <OP> <SF> <rD> <imm32>
	
	14h - D2F 	- 7B	  - <OP> <SF> <rS> <Addr32>
	15h - D2S	- 5B	  - <OP> <SF> <rS> <rB>:<rA>
	16h - D2R	- 4B	  - <OP> <SF> <rS> <rA>
	17h - D2SD  - 6B	  - <OP> <SF> <rS> <rB>:<rA> <disp8>
	18h - D2RD  - 5B 	  - <OP> <SF> <rS> <rA> <disp8>

	19h - D3F 	- 7B	  - <OP> <SF> <rD> <Addr32>
	1Ah - D3S	- 5B	  - <OP> <SF> <rD> <rB>:<rA>
	2Bh - D3R	- 4B	  - <OP> <SF> <rD> <rA>
	2Ch - D3SD  - 6B	  - <OP> <SF> <rD> <rB>:<rA> <disp8>
	2Dh - D3RD  - 5B	  - <OP> <SF> <rD> <rA> <disp8>

	
Instructions:
	#OP #SF - #Name   - #InsType - #FlMask  - #Description
	
	00h     - NOP 	  - S0 		 - -------- - No operation
	01h     - HALT 	  - S0       - -------- - Halt clock
	02h     - WAIT    - S0       - -------- - Halt until IRQ
	
	03h 00h - ADD	  - A0       - ---ZOS-C - Addition
	03h 01h - ADD.b   - A1 		 - ---ZOS-C - Add immediate (byte)
	03h 02h - ADD.w   - A2 		 - ---ZOS-C - Add immediate (word)
	03h 03h - ADD.dw  - A3 		 - ---ZOS-C - Add immediate (dword)
	
	04h 00h - ADC     - A0 		 - ---ZOS-C - Addition with carry
	04h 01h - ADC.b   - A1 		 - ---ZOS-C - Add carry immediate (byte)
	04h 02h - ADC.w   - A2 		 - ---ZOS-C - Add carry immediate (word)
	04h 03h - ADC.dw  - A3 	     - ---ZOS-C - Add carry immediate (dword)
	
	05h 00h - SUB	  - A0       - ---ZOSB- - Subtraction
	05h 01h - SUB.b   - A1 		 - ---ZOSB- - Subtract immediate (byte)
	05h 02h - SUB.w   - A2 		 - ---ZOSB- - Subtract immediate (word)
	05h 03h - SUB.dw  - A3 		 - ---ZOSB- - Subtract immediate (dword)
	
	06h 00h - SBB	  - A0       - ---ZOSB- - Subtraction with borrow
	06h 01h - SBB.b   - A1 		 - ---ZOSB- - Subtract with borrow immediate (byte)
	06h 02h - SBB.w   - A2 		 - ---ZOSB- - Subtract with borrow immediate (word)
	06h 03h - SBB.dw  - A3 		 - ---ZOSB- - Subtract with borrow immediate (dword)
	
	07h 00h - MUL	  - A0       - ---ZOS-- - Multiply
	07h 01h - MUL.b   - A1 		 - ---ZOS-- - Multiply immediate (byte)
	07h 02h - MUL.w   - A2 		 - ---ZOS-- - Multiply immediate (word)
	07h 03h - MUL.dw  - A3 		 - ---ZOS-- - Multiply immediate (dword)
	
	08h 00h - IMUL	  - A0       - ---ZOS-- - Signed Multiply
	08h 01h - IMUL.b  - A1 		 - ---ZOS-- - Signed Multiply immediate (byte)
	08h 02h - IMUL.w  - A2 		 - ---ZOS-- - Signed Multiply immediate (word)
	08h 03h - IMUL.dw - A3 		 - ---ZOS-- - Signed Multiply immediate (dword)
	
	09h 00h - DIV	  - A0       - ---ZOS-- - Divide
	09h 01h - DIV.b   - A1 		 - ---ZOS-- - Divide immediate (byte)
	09h 02h - DIV.w   - A2 		 - ---ZOS-- - Divide immediate (word)
	09h 03h - DIV.dw  - A3 		 - ---ZOS-- - Divide immediate (dword)
	
	0Ah 00h - IDIV	  - A0       - ---ZOS-- - Signed Divide
	0Ah 01h - IDIV.b  - A1 		 - ---ZOS-- - Signed Divide immediate (byte)
	0Ah 02h - IDIV.w  - A2 		 - ---ZOS-- - Signed Divide immediate (word)
	0Ah 03h - IDIV.dw - A3 		 - ---ZOS-- - Signed Divide immediate (dword)
	
	0Bh     - NOT     - S3       - ---ZOS--
	0Ch     - INC     - S3       - ---ZOS-C - Increment by one
	0Dh     - DEC     - S3       - ---ZOSB- - Decrement by one
	
	0Eh     - PUSH    - S2 		 - -------- - Push to stack
	0Fh     - POP     - S1		 - -------- - Pop from stack
	
	10h     - CALL    - S7       - -------- - Call subroutine (relative)
	11h     - CALLR   - S2       - -------- - Call subroutine (register)
	12h     - CLABS   - S7       - -------- - Call subroutine (absolute)
	13h     - RET     - S0       - -------- - Return from subroutine
	
	14h     - LSL     - H0		 - ---ZOS-- - Logical shift left
	15h     - LSR     - H0		 - ---ZOS-- - Logical shift right
	16h     - ASR     - H0		 - ---ZOS-- - Arithmetical shift right
	17h     - ROL     - H0		 - ---ZOS-- - Rotate left
	18h     - ROR     - H0		 - ---ZOS-- - Rotate right
	
	19h     - AND     - A0       - ---ZOS--
	1Ah     - OR      - A0       - ---ZOS--
	1Bh     - XOR	  - A0 		 - ---ZOS-- 
	
	1Ch 00h - CMP     - A0 		 - LEGZOSB- - Compare
	1Ch 01h - CMP.b   - A1 		 - LEGZOSB- - Compare immediate (byte)
	1Ch 02h - CMP.w   - A2 		 - LEGZOSB- - Compare immediate (word)
	1Ch 03h - CMP.dw  - A3 		 - LEGZOSB- - Compare immediate (dword)
	
	1Dh 00h - ICMP    - A0 		 - LEGZOSB- - Signed Compare
	1Dh 01h - ICMP.b  - A1 		 - LEGZOSB- - Signed Compare immediate (byte)
	1Dh 02h - ICMP.w  - A2 		 - LEGZOSB- - Signed Compare immediate (word)
	1Dh 03h - ICMP.dw - A3 		 - LEGZOSB- - Signed Compare immediate (dword)
	
	1Eh     - JMP     - J2       - -------- - Branch (relative)
	1Fh     - JMPR    - J1       - -------- - Branch (register)
	20h     - JMA     - J0       - -------- - Branch (absolute)
	
	21h 00h - COPY    - D0       - -------- - Copy (reg-reg)
	21h 01h - LDI.B   - D1B      - -------- - Load immediate (byte)
	21h 02h - LDI.W   - D1W      - -------- - Load immediate (word)
	21h 03h - LDI.DW  - D1DW     - -------- - Load immediate (dword)
	
	22h 00h - STR.B   - D2F      - -------- - Store byte (Absolute; reg-mem)
	22h 01h - STR.B   - D2S      - -------- - Store byte (Segment; reg-mem)
	22h 02h - STR.B   - D2R      - -------- - Store byte (Register; reg-mem)
	22h 03h - STR.B   - D2SD     - -------- - Store byte (Segment-displacement; reg-mem)
	22h 04h - STR.B   - D2RD     - -------- - Store byte (Register-displacement; reg-mem)
	
	22h 05h - STR.W   - D2F      - -------- - Store word (Absolute; reg-mem)
	22h 06h - STR.W   - D2S      - -------- - Store word (Segment; reg-mem)
	22h 07h - STR.W   - D2R      - -------- - Store word (Register; reg-mem)
	22h 08h - STR.W   - D2SD     - -------- - Store word (Segment-displacement; reg-mem)
	22h 09h - STR.W   - D2RD     - -------- - Store word (Register-displacement; reg-mem)
	
	22h 0Ah - STR.DW  - D2F      - -------- - Store dword (Absolute; reg-mem)
	22h 0Bh - STR.DW  - D2S      - -------- - Store dword (Segment; reg-mem)
	22h 0Ch - STR.DW  - D2R      - -------- - Store dword (Register; reg-mem)
	22h 0Dh - STR.DW  - D2SD     - -------- - Store dword (Segment-displacement; reg-mem)
	22h 0Eh - STR.DW  - D2RD     - -------- - Store dword (Register-displacement; reg-mem)
	
	23h 00h - LOD.B   - D3F      - -------- - Load byte (Absolute; reg-mem)
	23h 01h - LOD.B   - D3S      - -------- - Load byte (Segment; reg-mem)
	23h 02h - LOD.B   - D3R      - -------- - Load byte (Register; reg-mem)
	23h 03h - LOD.B   - D3SD     - -------- - Load byte (Segment-displacement; reg-mem)
	23h 04h - LOD.B   - D3RD     - -------- - Load byte (Register-displacement; reg-mem)
	
	23h 05h - LOD.W   - D3F      - -------- - Load word (Absolute; reg-mem)
	23h 06h - LOD.W   - D3S      - -------- - Load word (Segment; reg-mem)
	23h 07h - LOD.W   - D3R      - -------- - Load word (Register; reg-mem)
	23h 08h - LOD.W   - D3SD     - -------- - Load word (Segment-displacement; reg-mem)
	23h 09h - LOD.W   - D3RD     - -------- - Load word (Register-displacement; reg-mem)
	
	23h 0Ah - LOD.DW  - D3F      - -------- - Load dword (Absolute; reg-mem)
	23h 0Bh - LOD.DW  - D3S      - -------- - Load dword (Segment; reg-mem)
	23h 0Ch - LOD.DW  - D3R      - -------- - Load dword (Register; reg-mem)
	23h 0Dh - LOD.DW  - D3SD     - -------- - Load dword (Segment-displacement; reg-mem)
	23h 0Eh - LOD.DW  - D3RD     - -------- - Load dword (Register-displacement; reg-mem)
	
	24h     - WRINT   - I0       - -------- - Write interrupt
	25h     - INT     - S4       - -------- - Interrupt
	26h     - IRET    - S0       - -------- - Return from interrupt
	27h     - CLI     - S0       - -------- - Disable interrupts
	28h     - STI     - S0       - -------- - Enable interrupts
	
	29h 00h - REM     - A0       - ---ZOS-- - Remainder
	29h 01h - REM.b   - A1       - ---ZOS-- - Remainder immediate (byte)
	29h 02h - REM.w   - A2       - ---ZOS-- - Remainder immediate (word)
	29h 03h - REM.dw  - A3       - ---ZOS-- - Remainder immediate (dword)
	
	2Ah 00h - IREM    - A0       - ---ZOS-- - Signed Remainder
	2Ah 01h - IREM.b  - A1       - ---ZOS-- - Signed Remainder immediate (byte)
	2Ah 02h - IREM.w  - A2       - ---ZOS-- - Signed Remainder immediate (word)
	2Ah 03h - IREM.dw - A3       - ---ZOS-- - Signed Remainder immediate (dword)
