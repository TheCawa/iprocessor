Specs:
	CPU - i80148
	RAM - 16M (up to 4096M)
	
Memory address map:
	BANK 0000h:
	#START    - #END      - #NAME   
	
	00000000h - 0000FFFFh - BIOS/VBIOS ROM
	00010000h - 0001FFFFh - BIOS RAM
	00020000h - 0002FFFFh - MMIO
	00030000h - FFFFFFFFh - USER RAM
	
	BANK 0001h:
	00000000h-FFFFFFFFh - Additional data
	
	BANK 0002h-FFFFh - Other
	Optional:
	BANK 0002h - VRAM Mirror (up to 4GB)
	BANK 0003h - Audio Buffer Mirror (up to 4GB)
	BANK 0004h - Ethernet buffer (Example: Network stack or other buffers; up to 4 GB)
	BANK FF00h-FFFFh - XPB buffers
