Specs:
	CPU - i80148
	RAM - 16M (up to 4096M)
	
Memory regions:
	BANK 0000h:
	#START    - #END      - #NAME   
	
	00000000h - 0000FFFFh - BIOS/VBIOS ROM
	00010000h - 0001FFFFh - CMOS
	00020000h - 0002FFFFh - MMIO
	00030000h - 0003FFFFh - DEVICE CALLS
	00040000h - 0004FFFFh - BIOS RAM
	00050000h - FFFFFFFFh - USER RAM
	
	-# IMPORTANT! Regions 00030000h-0003FFFFh, 00040000h-0004FFFFh and 00050000h-FFFFFFFFh it's more like it's just formalized!
	
	BANK 0001h - Additional data
	BANK 0002h - Additional data
	BANK 0003h-FEFFh - Other
	BANK FF00h-FFFFh - XPB Banks

XPB PROTOCOLS:
	Device registers (ports) map protocol:
		#ID - #AC - #NAME 		- #Description
		
		00h - R   - DEV_DATA    - Device class and vendor
		01h - X   - Reserved
		02h - R/W - DEV_STATUS  - Device status
		03h - R   - FEATURES    - Device features
		04h - X   - Reserved
		05h - X   - Reserved
		06h - X   - Reserved
		07h - X   - Reserved
		08h - X   - DSR
		09h - X   - DSR
		0Ah - X   - DSR
		0Bh - X   - DSR
		0Ch - X   - DSR
		0Dh - X   - DSR
		0Eh - X   - DSR
		0Fh - X   - DSR
		
		-# DSR - Device service registers
	
IRQ:
